package es.us.dp1.lx_xy_24_25.dixit.game;

import es.us.dp1.lx_xy_24_25.dixit.card.Card;
import es.us.dp1.lx_xy_24_25.dixit.player.Player;
import org.springframework.web.bind.annotation.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;

@RestController
@RequestMapping("/api/v1/turns")
public class TurnController {
    private final TurnService turnService;

    @Autowired
    public TurnController(TurnService turnService) {
        this.turnService = turnService;
    }

    @PutMapping("/{turnId}/answer/{cardId}")
    public ResponseEntity<Turn> setStorytellerAnswer(
        @PathVariable Integer turnId,
        @PathVariable Integer cardId
    ) {
        Turn turn = turnService.findById(turnId);
        Card card = turnService.findCardById(cardId);
        Turn updatedTurn = turnService.setStorytellerAnswer(turn, card);
        return ResponseEntity.ok(updatedTurn);
    }

    @PutMapping("/{turnId}/clue")
    public ResponseEntity<Turn> setStorytellerClue(
        @PathVariable Integer turnId,
        @RequestBody String clue
    ) {
        Turn turn = turnService.findById(turnId);
        Turn updatedTurn = turnService.setStorytellerClue(turn, clue);
        return ResponseEntity.ok(updatedTurn);
    }

    @PostMapping("/{turnId}/play-card/{cardId}/{playerId}")
    public ResponseEntity<Card> playerPlayCard(
        @PathVariable Integer turnId,
        @PathVariable Integer cardId,
        @PathVariable Integer playerId
    ) {
        Turn turn = turnService.findById(turnId);
        Player player = turnService.findPlayerById(playerId);
        Card card = turnService.findCardById(cardId);
        Card playedCard = turnService.playerPlayCard(turn, player, card);
        return ResponseEntity.ok(playedCard);
    }

    @PostMapping("/{turnId}/vote/{playerId}/{cardId}")
    public ResponseEntity<Player> playerVote(
        @PathVariable Integer turnId,
        @PathVariable Integer playerId,
        @PathVariable Integer cardId
    ) {
        Turn turn = turnService.findById(turnId);
        Player player = turnService.findPlayerById(playerId);
        Card card = turnService.findCardById(cardId);
        Player updatedPlayer = turnService.playerVote(turn, player, card);
        return ResponseEntity.ok(updatedPlayer);
    }
}
