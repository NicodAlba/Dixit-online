package es.us.dp1.lx_xy_24_25.dixit.achievement;

import es.us.dp1.lx_xy_24_25.dixit.auth.payload.response.MessageResponse;
import es.us.dp1.lx_xy_24_25.dixit.util.RestPreconditions;
import io.swagger.v3.oas.annotations.parameters.RequestBody;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/v1/achievements")
@Tag(name = "Achievements")
public class AchievementRestController {
    private final AchievementService achievementService;

    @Autowired
    public AchievementRestController(AchievementService achievementService){
        this.achievementService = achievementService;
    }

    @GetMapping
    public ResponseEntity<List<Achievement>> getAllAchievements(@RequestParam(required = false) Integer threshold,
                                                                @RequestParam(required = false) Condition condition) {
        List<Achievement> res;
        if (threshold != null){
            res = achievementService.findByThreshold(threshold);
        } else if (condition != null){
            res = achievementService.findByCondition(condition);
        } else {
            res = achievementService.findAll();
        }
        return new ResponseEntity<>(res, HttpStatus.OK);
    }

    @GetMapping("/{id}")
    public ResponseEntity<Achievement> getAchievement(@PathVariable Integer id){
        Achievement a = achievementService.findById(id);
        return new ResponseEntity<>(a, HttpStatus.OK);
    }

    @PostMapping
    @ResponseStatus(HttpStatus.CREATED)
    public ResponseEntity<Achievement> createAchievement(@RequestBody @Valid Achievement achievement){
        Achievement createdAchievement = achievementService.saveAchievement(achievement);
        return new ResponseEntity<>(createdAchievement, HttpStatus.CREATED);
    }

    @PutMapping("/{id}")
    @ResponseStatus(HttpStatus.OK)
    public ResponseEntity<Achievement> updateAchievement(@RequestBody @Valid Achievement achievement, @PathVariable Integer id){
        RestPreconditions.checkNotNull(achievementService.findById(id), "Achievement", "id", id);
        Achievement updated = achievementService.updateAchievement(achievement, id);
        return new ResponseEntity<>(updated, HttpStatus.OK);
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.OK)
    public ResponseEntity<MessageResponse> deleteAchievement(Integer id){
        RestPreconditions.checkNotNull(achievementService.findById(id), "Achievement", "id", id);
        achievementService.deleteAchievement(id);
        return new ResponseEntity<>(new MessageResponse("Achievement deleted!"), HttpStatus.OK);
    }
}
